    17→pub fn attack_overrides(
    18→    game: &GameState,
    19→    _attack: &Attack,
    20→    attacker_id: CardInstanceId,
    21→    defender_id: CardInstanceId,
    22→) -> AttackOverrides {
    23→    let mut overrides = AttackOverrides::default();
    24→    let attacker = match game.current_player().find_pokemon(attacker_id) {
    25→        Some(slot) => slot,
    26→        None => return overrides,
    27→    };
    28→    let defender = match game.opponent_player().find_pokemon(defender_id) {
    29→        Some(slot) => slot,
    30→        None => return overrides,
    31→    };
    32→
    33→    if game.pokemon_is_delta(attacker_id) && player_has_feraligatr_battle_aura(game, attacker_id) {
    34→        overrides.pre_weakness_modifier += 10;
    35→    }
    36→
    37→    if holon_legacy_in_play(game) && game.pokemon_is_delta(defender_id) {
    38→        overrides.ignore_weakness = true;
    39→    }
    40→
    41→    if defender.is_ex == false
    42→        && has_holon_energy(game, defender, 84)
    43→        && has_basic_energy_of_type(game, defender, Type::Fire)
    44→    {
    45→        overrides.ignore_weakness = true;
    46→    }
    47→
    48→    if attacker.is_ex == false
    49→        && has_holon_energy(game, attacker, 84)
    50→        && has_basic_energy_of_type(game, attacker, Type::Fighting)
    51→    {
    52→        overrides.ignore_resistance = true;
    53→    }
    54→
    55→    if defender.is_ex == false
    56→        && attacker.is_ex
    57→        && has_holon_energy(game, defender, 85)
    58→        && has_basic_energy_of_type(game, defender, Type::Lightning)
    59→    {
    60→        overrides.damage_modifier -= 10;
    61→    }
    62→
    63→    // Heracross δ (DF-3) Shining Horn:
    64→    // If Heracross is Active and has no Benched Pokémon, prevent all damage done to it by Basic Pokémon.
    65→    if def_id_matches(&defender.card.def_id, "DF", 3)
    66→        && attacker.stage == Stage::Basic
    67→        && game.opponent_player().bench.is_empty()
    68→        && body_active(game, defender.card.id)
    69→    {
    70→        overrides.prevent_damage = true;
    71→    }
    72→
    73→    if def_id_matches(&defender.card.def_id, "DF", 9)
    74→        && game.current_player().hand.count() >= 5
    75→        && body_active(game, defender.card.id)
    76→    {
    77→        overrides.damage_modifier -= 30;
    78→    }
    79→
    80→    if def_id_matches(&defender.card.def_id, "DF", 15)
    81→        && game.pokemon_is_delta(attacker_id)
    82→        && body_active(game, defender.card.id)
    83→    {
    84→        overrides.damage_modifier -= 40;
    85→    }
    86→
    87→    if defender.is_ex
    88→        && defender.stage == Stage::Stage2
    89→        && player_has_kingdra_extra_smoke(game, defender_id)
    90→    {
    91→        overrides.pre_weakness_modifier -= 10;
    92→    }
    93→
    94→    if attacker.is_ex
    95→        && attacker.stage == Stage::Stage2
    96→        && player_has_xatu_extra_feather(game, attacker_id)
    97→    {
    98→        overrides.pre_weakness_modifier += 10;
    99→    }
   100→
   101→    if def_id_matches(&attacker.card.def_id, "DF", 1) && _attack.name == "Delta Circle" {
   102→        let count = count_delta_pokemon_in_play(game, attacker_id);
   103→        if count > 0 {
   104→            overrides.pre_weakness_modifier += (count as i32) * 10;
   105→        }
   106→    }
   107→
   108→    if def_id_matches(&attacker.card.def_id, "DF", 3)
   109→        && _attack.name == "Extra Claws"
   110→        && defender.is_ex
   111→    {
   112→        overrides.pre_weakness_modifier += 20;
   113→    }
   114→
   115→    // DF-7 Nidoqueen: Vengeance attack
   116→    if def_id_matches(&attacker.card.def_id, "DF", 7) && _attack.name == "Vengeance" {
   117→        overrides.pre_weakness_modifier += df_007_nidoqueen::vengeance_bonus(game, attacker_id);
   118→    }
   119→
   120→    if def_id_matches(&attacker.card.def_id, "DF", 15)
   121→        && _attack.name == "Surge"
   122→        && count_energy_of_type(game, attacker, Type::Water) >= 2
   123→    {
   124→        overrides.pre_weakness_modifier += 20;
   125→    }
   126→
   127→    if def_id_matches(&attacker.card.def_id, "DF", 16)
   128→        && _attack.name == "Tail Sting"
   129→        && defender.is_ex
   130→    {
   131→        overrides.pre_weakness_modifier += 10;
   132→    }
   133→
   134→    if def_id_matches(&attacker.card.def_id, "DF", 21)
   135→        && _attack.name == "Pump Out"
   136→        && has_any_tool(attacker)
   137→    {
   138→        overrides.pre_weakness_modifier += 20;
   139→    }
   140→
   141→    if def_id_matches(&attacker.card.def_id, "DF", 23) && _attack.name == "Grind" {
   142→        let total = count_total_energy(attacker) as i32 * 10;
   143→        let modifier = total - _attack.damage as i32;
   144→        if modifier != 0 {
   145→            overrides.pre_weakness_modifier += modifier;
   146→        }
   147→    }
   148→
   149→    if def_id_matches(&attacker.card.def_id, "DF", 98)
   150→        && _attack.name == "Dual Stream"
   151→        && game.has_marker(attacker_id, "Dual Stream Reduced")
   152→    {
   153→        overrides.pre_weakness_modifier -= 40;
   154→    }
   155→
   156→    if def_id_matches(&attacker.card.def_id, "DF", 32) && _attack.name == "Link Blast" {
   157→        let attacker_energy = count_total_energy(attacker);
   158→        let defender_energy = count_total_energy(defender);
   159→        if attacker_energy != defender_energy {
   160→            overrides.pre_weakness_modifier -= 30;
   161→        }
   162→    }
   163→
   164→    if def_id_matches(&attacker.card.def_id, "DF", 35) && _attack.name == "Rage" {
   165→        let bonus = attacker.damage_counters as i32 * 10;
   166→        if bonus > 0 {
   167→            overrides.pre_weakness_modifier += bonus;
   168→        }
   169→    }
   170→
   171→    if def_id_matches(&attacker.card.def_id, "DF", 37)
   172→        && _attack.name == "Extra Ball"
   173→        && defender.is_ex
   174→    {
   175→        overrides.pre_weakness_modifier += 30;
   176→    }
   177→
   178→    if def_id_matches(&attacker.card.def_id, "DF", 39)
   179→        && _attack.name == "Flickering Tail"
   180→        && defender.is_ex
   181→    {
   182→        overrides.pre_weakness_modifier += 10;
   183→    }
   184→
   185→    if def_id_matches(&attacker.card.def_id, "DF", 49) && _attack.name == "Flail" {
   186→        let bonus = attacker.damage_counters as i32 * 10;
   187→        if bonus > 0 {
   188→            overrides.pre_weakness_modifier += bonus;
   189→        }
   190→    }
   191→
   192→    if def_id_matches(&attacker.card.def_id, "DF", 60) && _attack.name == "Psychic Boom" {
   193→        let bonus = defender.attached_energy.len() as i32 * 10;
   194→        if bonus > 0 {
   195→            overrides.pre_weakness_modifier += bonus;
   196→        }
   197→    }
   198→
   199→    if def_id_matches(&attacker.card.def_id, "DF", 65) && _attack.name == "Splash About" {
   200→        let attacker_energy = count_total_energy(attacker);
   201→        let defender_energy = count_total_energy(defender);
   202→        if attacker_energy < defender_energy {
   203→            overrides.pre_weakness_modifier += 10;
   204→        }
   205→    }
   206→
   207→    if def_id_matches(&attacker.card.def_id, "DF", 99)
   208→        && _attack.name == "Hyper Claws"
   209→        && defender.stage == Stage::Stage2
   210→    {
   211→        overrides.pre_weakness_modifier += 20;
   212→    }
   213→
   214→    if game.has_marker(attacker_id, "Delta Reduction") {
   215→        overrides.pre_weakness_modifier -= 30;
   216→    }
   217→
   218→    if game.has_marker(attacker_id, "Charm") {
   219→        overrides.pre_weakness_modifier -= 20;
   220→    }
   221→
   222→    if game.has_marker(defender_id, "Granite Head") {
   223→        overrides.damage_modifier -= 10;
   224→    }
   225→
   226→    if game.has_marker(defender_id, "Protective Swirl") {
   227→        overrides.ignore_weakness = true;
   228→    }
   229→
   230→    overrides
   231→}

   234→pub fn attack_blocked_by_shining_horn(game: &GameState, attacker_id: CardInstanceId) -> bool {
   235→    let attacker = match game.current_player().find_pokemon(attacker_id) {
   236→        Some(slot) => slot,
   237→        None => return false,
   238→    };
   239→    if attacker.stage != Stage::Basic {
   240→        return false;
   241→    }
   242→    let opponent = game.opponent_player();
   243→    let active = match opponent.active.as_ref() {
   244→        Some(slot) => slot,
   245→        None => return false,
   246→    };
   247→    if !def_id_matches(&active.card.def_id, "DF", 3) {
   248→        return false;
   249→    }
   250→    opponent.bench.is_empty()
   251→}

   261→pub fn between_turns(game: &mut GameState) {
   262→    let sleeping_snorlax = [
   263→        (&game.players[0], PlayerId::P1),
   264→        (&game.players[1], PlayerId::P2),
   265→    ]
   266→    .into_iter()
   267→    .filter_map(|(player, id)| {
   268→        let active = player.active.as_ref()?;
   269→        if def_id_matches(&active.card.def_id, "DF", 10)
   270→            && active.has_special_condition(SpecialCondition::Asleep)
   271→        {
   272→            Some(id)
   273→        } else {
   274→            None
   275→        }
   276→    })
   277→    .collect::<Vec<_>>();
   278→
   279→    for sleeping_owner in sleeping_snorlax {
   280→        let opponent = match sleeping_owner {
   281→            PlayerId::P1 => PlayerId::P2,
   282→            PlayerId::P2 => PlayerId::P1,
   283→        };
   284→        game.deal_between_turns_damage(opponent, 20);
   285→    }
   286→
   287→    discard_boost_energy(game);
   288→}

   417→pub fn energy_provides_override(_game: &GameState, _card: &CardInstance) -> Option<Vec<Type>> {
   418→    if def_id_matches(&_card.def_id, "DF", 88) {
   419→        if let Some(slot) = attached_slot_for_energy(_game, _card.id) {
   420→            if slot.is_delta {
   421→                return Some(all_energy_types());
   422→            }
   423→        }
   424→        return Some(vec![Type::Colorless]);
   425→    }
   426→    if def_id_matches(&_card.def_id, "DF", 89) {
   427→        if let Some(slot) = attached_slot_for_energy(_game, _card.id) {
   428→            if scramble_energy_active(_game, slot) {
   429→                return Some(all_energy_types());
   430→            }
   431→        }
   432→        return Some(vec![Type::Colorless]);
   433→    }
   434→    if def_id_matches(&_card.def_id, "DF", 87) {
   435→        return Some(vec![Type::Colorless]);
   436→    }
   437→    None
   438→}

   440→pub fn on_energy_attached(
   441→    game: &mut GameState,
   442→    energy_id: CardInstanceId,
   443→    target_id: CardInstanceId,
   444→    _source: EnergyAttachmentSource,
   445→) {
   446→    let slot = match game
   447→        .players
   448→        .iter_mut()
   449→        .find_map(|player| player.find_pokemon_mut(target_id))
   450→    {
   451→        Some(slot) => slot,
   452→        None => return,
   453→    };
   454→    let energy_def_id = slot
   455→        .attached_energy
   456→        .iter()
   457→        .find(|energy| energy.id == energy_id)
   458→        .map(|energy| energy.def_id.clone());
   459→    let Some(energy_def_id) = energy_def_id else {
   460→        return;
   461→    };
   462→    if def_id_matches(&energy_def_id, "DF", 87) {
   463→        if slot.stage == Stage::Basic {
   464→            discard_attached_energy(game, target_id, energy_id);
   465→            return;
   466→        }
   467→        let mut marker = Marker::new("Boost Energy");
   468→        marker.source = Some(energy_id);
   469→        marker.expires_after_turn = Some(game.turn.number);
   470→        slot.markers.push(marker);
   471→    }
   472→    if def_id_matches(&energy_def_id, "DF", 89) {
   473→        if slot.stage == Stage::Basic || slot.is_ex {
   474→            discard_attached_energy(game, target_id, energy_id);
   475→        }
   476→    }
   477→}

   627→fn execute_copycat(game: &mut GameState, source_id: CardInstanceId) -> bool {
   628→    let owner = match owner_for_source(game, source_id) {
   629→        Some(player) => player,

   633→        PlayerId::P1 => 0,
   634→        PlayerId::P2 => 1,
   635→    };
   636→    let opponent_index = 1 - owner_index;
   637→    let opponent_hand_count = game.players[opponent_index].hand.count();
   638→    let hand_ids = game.players[owner_index].hand.order();
   639→    for card_id in hand_ids {

   642→        }
   643→    }
   644→    let seed = game.rng.next_u64();
   645→    game.players[owner_index].shuffle_deck(seed);
   646→    let _ = game.draw_cards_with_events(owner, opponent_hand_count);
   647→    true
   648→}

   698→fn resolve_holon_mentor_discard(
   699→    game: &mut GameState,
   700→    source_id: Option<CardInstanceId>,
   701→    target_ids: &[CardInstanceId],
   702→) -> bool {
   703→    let source_id = match source_id {
   704→        Some(id) => id,
   705→        None => return false,
   706→    };
   707→    if target_ids.len() != 1 {
   708→        return false;
   709→    }
   710→    let owner = match owner_for_source(game, source_id) {
   711→        Some(player) => player,
   712→        None => return false,
   713→    };
   714→    let owner_index = match owner {
   715→        PlayerId::P1 => 0,
   716→        PlayerId::P2 => 1,
   717→    };
   718→    if let Some(card) = game.players[owner_index].hand.remove(target_ids[0]) {
   719→        game.players[owner_index].discard.add(card);
   720→    } else {
   721→        return false;
   722→    }
   723→
   724→    let matching_cards: Vec<_> = game.players[owner_index]
   725→        .deck
   726→        .cards()
   727→        .iter()
   728→        .filter(|card| {
   729→            game.card_meta
   730→                .get(&card.def_id)
   731→                .map(|meta| meta.is_pokemon && meta.is_basic && meta.hp <= 100)
   732→                .unwrap_or(false)
   733→        })
   734→        .cloned()
   735→        .collect();
   736→    if matching_cards.is_empty() {
   737→        return true;
   738→    }
   739→    let options: Vec<_> = matching_cards.iter().map(|card| card.id).collect();
   740→    let revealed_cards = matching_cards
   741→        .iter()
   742→        .map(|card| tcg_core::RevealedCard {
   743→            id: card.id,
   744→            def_id: card.def_id.as_str().to_string(),
   745→            name: game
   746→                .card_meta
   747→                .get(&card.def_id)
   748→                .map(|meta| meta.name.clone())
   749→                .unwrap_or_else(|| card.def_id.as_str().to_string()),
   750→        })
   751→        .collect();
   752→    let prompt = Prompt::ChooseCardsFromDeck {
   753→        player: owner,
   754→        count: 3,
   755→        options,
   756→        revealed_cards,
   757→        min: Some(0),
   758→        max: Some(3),
   759→        destination: SelectionDestination::Hand,
   760→        shuffle: true,
   761→    };
   762→    game.set_pending_prompt_custom(prompt, owner, "DF-75:Holon Mentor:Deck".to_string(), Some(source_id));
   763→    true
   764→}

  1278→fn perform_copied_attack(
  1279→    game: &mut GameState,
  1280→    source_id: CardInstanceId,
  1281→    attack: &Attack,
  1282→) -> bool {
  1283→    if !game.attack_cost_met(source_id, attack) {
  1284→        return true;
  1285→    }
  1286→    let defender_id = match game.opponent_player().active.as_ref() {
  1287→        Some(slot) => slot.card.id,
  1288→        None => return false,
  1289→    };
  1290→    let defender = match game.opponent_player().active.as_ref() {
  1291→        Some(slot) => slot.clone(),
  1292→        None => return false,
  1293→    };
  1294→    let mut attack_override = attack.clone();
  1295→    attack_override.attack_type = game.attack_type_for(source_id, &attack_override);
  1296→    let overrides = tcg_core::apply_attack_overrides(
  1297→        game,
  1298→        &attack_override,
  1299→        source_id,
  1300→        defender_id,
  1301→    );
  1302→    if !overrides.prevent_damage && overrides.damage_modifier != 0 {
  1303→        game.add_damage_modifier(overrides.damage_modifier);
  1304→    }
  1305→    if overrides.pre_weakness_modifier != 0 {
  1306→        let adjusted = attack_override.damage as i32 + overrides.pre_weakness_modifier;
  1307→        attack_override.damage = adjusted.max(0) as u16;
  1308→    }
  1309→    let base_damage = tcg_core::calculate_damage_with_flags(
  1310→        &attack_override,
  1311→        &defender,
  1312→        overrides.ignore_weakness,
  1313→        overrides.ignore_resistance,
  1314→    );
  1315→    let post_weakness = game.damage_after_weakness_modifier(source_id, defender_id);
  1316→    let adjusted = (base_damage as i32 + post_weakness).max(0) as u16;
  1317→    let damage = tcg_core::apply_damage_modifier(
  1318→        adjusted,
  1319→        game.peek_attack_damage_modifier(source_id, defender_id),
  1320→    );
  1321→    let manual_damage = match attack_override.effect_ast.as_ref() {
  1322→        Some(EffectAst::DealDamageNoModifiers { amount, .. }) => Some(*amount as u16),
  1323→        Some(EffectAst::DealDamageNoModifiersWithOverkillRecoil { amount, .. }) => {
  1324→            Some(*amount as u16)
  1325→        }
  1326→        _ => None,
  1327→    };
  1328→    if !overrides.prevent_damage {
  1329→        if manual_damage.is_none() {
  1330→            let _ = tcg_core::execute_attack(game, &attack_override);
  1331→        } else if let Some(effect) = attack_override.effect_ast.as_ref() {
  1332→            let _ = tcg_core::execute_effect_with_source(game, effect, Some(source_id));
  1333→        }
  1334→        if manual_damage.is_none() {
  1335→            if !game.prevents_attack_effects(source_id, defender_id) {
  1336→                if let Some(effect) = attack_override.effect_ast.as_ref() {
  1337→                    let _ = tcg_core::execute_effect_with_source(game, effect, Some(source_id));
  1338→                }
  1339→            }
  1340→        }
  1341→        tcg_core::apply_post_attack_custom(
  1342→            game,
  1343→            source_id,
  1344→            defender_id,
  1345→            damage,
  1346→        );
  1347→    }
  1348→    tcg_core::after_attack(game, source_id, defender_id);
  1349→    true
  1350→}
